-- lower id kedudukan 
SELECT 
    dn.nama,
    dn.kedudukan,
    k.id_kedudukan,
    k.nama_kedudukan
FROM data_notaris dn
JOIN kedudukan k 
  ON TRIM(LOWER(REPLACE(dn.kedudukan, '  ', ' '))) = TRIM(LOWER(k.nama_kedudukan))
WHERE dn.id_kedudukan IS NULL;


-- update 

UPDATE data_notaris dn
JOIN kedudukan k 
  ON TRIM(LOWER(REPLACE(dn.kedudukan, '  ', ' '))) = TRIM(LOWER(k.nama_kedudukan))
SET dn.id_kedudukan = k.id_kedudukan
WHERE dn.id_kedudukan IS NULL;
